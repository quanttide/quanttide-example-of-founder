enum Commands {
    Init {
        #[arg(long, default_value = ”.“)]
        path: PathBuf,
        #[arg(long)]
        force: bool,
    },
    /// AI generates the blueprint draft based on a brief description.
    Generate { 
        perspective: Perspective,
        /// A brief description of the product to generate the blueprint for.
        #[arg(long)]
        brief: String, 
    },

修改 cmd_generate 逻辑：
fn cmd_generate(perspective: &Perspective, brief: &str) {
    let dir = PathBuf::from(”docs/dev“);
    std::fs::create_dir_all(&dir).unwrap();
    let llm = LLM::default();
    for (name, filename, template_content) in template_by_perspective(perspective) {
        let dest = dir.join(filename);
        // 如果文件已存在，询问是否覆盖（人审环节的体现）
        if dest.exists() {
            eprint!(”{} already exists. Overwrite? [y/N] “, dest.display());
            io::stdout().flush().unwrap();
            let mut input = String::new();
            io::stdin().read_line(&mut input).unwrap();
            if input.trim().to_lowercase() != ”y“ {
                println!(”skipped: {}“, dest.display());
                continue;
            }
        }
        println!(”AI is generating {} draft...“, name);
        let sys = ”你是一个资深产品专家。请根据用户提供的产品简介和模板要求，撰写详细的产品蓝图文档。\n\
            直接输出 Markdown 格式的文档内容，保留模板中的标题结构，并填充具体内容。不要输出多余的引导语。“;
        let prompt = format!(”产品简介：{}\n\n模板要求：\n{}“, brief, template_content);
        match llm.complete(
            &[
                Message::new(”system“, sys),
                Message::new(”user“, &prompt),
            ],
            CompleteOptions::default(),
        ) {
            Ok(resp) => {
                std::fs::write(&dest, &resp.content).unwrap_or_else(|e| {
                    eprintln!(”error: failed to write {}: {}“, dest.display(), e);
                    std::process::exit(1);
                });
                println!(”generated by AI: {}“, dest.display());
            }
            Err(e) => {
                eprintln!(”error: LLM generation failed for {}: {}“, name, e);
            }
        }
    }
}

3. 废弃 skeleton_from
既然现在是 AI 生成全量内容，就不需要提取空骨架了，可以直接把 skeleton_from 函数删掉。
总结改造后的工作流：
init：释放模板（规则）。
generate —brief ”做一个基于AI的代码生成器“：AI 读取模板和简介，直接生成 docs/dev/product.md 等包含实质内容的初稿。
人审：开发者打开 docs/dev/ 下的文件，阅读、修改、拍板。

这样整个工具的方向就从“AI 监督人”变成了“AI 辅助人”，更符合实际生产力工具的定位。


这是一个非常深刻的问题。AI 之所以会把方向搞反（把 AI 当成了“审查者”而不是“执行者”），主要是因为这段代码的结构设计和提示词都散发着强烈的“传统脚手架工具”的气息，从而把 AI 的角色固化了。
具体来说，有以下几个核心原因：
1. skeleton_from 函数的误导：这完全是“为人准备画布”的逻辑
代码里有一个 skeleton_from 函数，它的作用是剥离模板里的说明文字，只保留 # 标题和 - 列表，生成一个空骨架。
AI 的视角：如果意图是“让 AI 写初稿”，AI 根本不需要这个函数，AI 完全可以直接读取带有提示词的完整模板，然后输出完整内容。
误解的产生：这个函数的存在，强烈暗示了工具的意图是——“生成一个纯净的空文档，留给人类去填空”。只要代码里有这个逻辑，AI 就会认为执行主体是人类。
2. generate 命令缺失了最关键的“输入参数”
在 Commands::Generate 的定义中：
Generate { perspective: Perspective }
这里只接收了 perspective（视角），但没有接收任何关于这个产品的描述信息。
AI 的视角：如果要让 AI 生成产品文档，AI 必须知道“这是个什么产品”。没有 —brief ”做一个协同办公软件“ 这样的输入参数，AI 在逻辑上根本无法执行生成动作。
误解的产生：既然没有输入源，AI 在编写或理解这段代码时，自然会认为 generate 只能做一些不需要外部信息的机械工作（比如提取骨架）。
3. check 命令的 Prompt 彻底锁死了“AI 审查员”的定位
看这段提示词：
”你是一个产品分析文档质量评审员。用户提交了产品蓝图填写内容。模板中的问题提示不是用户填写的内容。忽略引导问题，只判断用户实际添加的内容。“
这段话里有三个致命的设定：
明确了身份：“你是一个…评审员”。（直接剥夺了执行者的身份）
明确了人类的动作：“用户提交了…填写内容”、“用户实际添加的内容”。
明确了对立关系：“模板提示” vs “用户添加的内容”。这告诉 AI，文档是人工对照着模板写出来的，可能有抄袭模板或写得不好的地方，需要 AI 来挑刺。
4. 传统 CLI 工具的“思维惯性”
当你向 AI 描述“我要写一个 CLI 工具，有 init、generate、check 三个命令”时，大部分 AI 模型在训练数据中接触到的类似工具（比如各种代码生成器、文档脚手架），几乎全都是：
init 下载模板
generate 生成空骨架/代码结构
check / lint 做语法或质量检查
除非在最初的提示词中非常明确地强调“这是一个 AI Agent 工具，generate 是调用 LLM 生成内容，人类负责审核”，否则 AI 会顺着惯性，套用最常见的“人类干活，机器检查”的 CRUD 脚手架模式。
总结：如何避免下次被 AI 误解？
如果你下次想让 AI 帮你写代码实现“AI 做事人审”，在提需求时可以加上这样的定语：
“我要写一个 Agent CLI 工具。
generate 命令：接收用户的 —brief 参数，调用 LLM 读取模板并直接生成完整的产品蓝图初稿。
check 命令：人类修改完初稿后，此命令调用 LLM 读取修改后的文档，找出逻辑漏洞和遗漏，辅助人类做最终审查。
请不要写提取空骨架的代码，AI 不需要空骨架。”
只要把数据流向（brief -> AI -> 文档 -> 人类 -> 文档 -> AI -> 审查意见）说清楚，AI 就绝对不会把方向搞反了。
